\begin{lemma}
  Let $\nabla$ be an affine connection on $\mathbb{R}^n$ and denote by $\omega$ its connection $1$-form in the standard frame. For $p, v \in \mathbb{R}^n$, we have
  \begin{equation}
    \Gamma(p \to p + v) = \Big[\mathbf{1} - \omega_p(v)\Big] + O_p(v^2).
  \end{equation}
  as $v \to 0$.
\end{lemma}

\begin{lemma}
  Let $\nabla$ be an affine connection on $\mathbb{R}^n$ and denote by $\omega$ its connection $1$-form in the standard frame. For $p, v \in \mathbb{R}^n$, we have
  \begin{equation}
    \tau_\nabla(p \to p + v) = v + O_p(v^2).
  \end{equation}
  as $v \to 0$.
\end{lemma}
