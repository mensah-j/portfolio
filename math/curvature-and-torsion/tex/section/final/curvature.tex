\section{Principal and linear connections}

\begin{definition}
  Let $M$ be a smooth manifold and let $E$ be a smooth fiber bundle over $M$ with Ehresmann connection $H \oplus V = \mathrm{T}E$. The \emph{curvature} of the connection is the map $\Omega \colon \Gamma(\mathrm{T}E) \times \Gamma(\mathrm{T}E) \to \Gamma(\mathrm{T}E)$ defined by 
  \begin{equation}
    \Omega^E(X, Y) = [X_H, Y_H]_V.
  \end{equation}
\end{definition}

The curvature defines a differential form; indeed, for a smooth function $f \in C^\infty(P)$, we have
\begin{align}
  \Omega^E(X, fY) = [X_H, fY_H]_V
  &= f[X_H, Y_H]_V + \big[(X_H f) Y_H\big]_V \\
  &= f[X_H, Y_H]_V + (X_H f) \big[Y_H\big]_V \\
  &= f[X_H, Y_H]_V,
\end{align}
which is all that is needed to check by antisymmetry and additivity of the Lie bracket. In special cases, one may define a related curvature form on the base manifold $M$.

\subsection{Curvature of principal connections}
Given a principal $G$-bundle $\rho \colon P \to M$, define the \emph{adjoint bundle} to be the associated bundle $\operatorname{ad} P = P \times_{\mathrm{Ad}} \mathfrak{g}$, where $\mathrm{Ad}$ is the adjoint action of $G$ on $\mathfrak{g}$. Explicitly, we have
\begin{equation}
  \operatorname{ad} P = (P \times \mathfrak{g}) / {\sim}; \quad \quad [u, v] \sim [u \cdot g, \mathrm{Ad}_g(v)].
\end{equation}

\end{equation}

  \Omega^P(g_* X, g_* Y) = [g_* X_H, g_* Y_H]_V = g_*[X_H, Y_H]_V = g_*\Omega^P(X, Y),
\end{equation}
since $g_*$ preserves the decomposition $\mathrm{T}P = H \oplus V$ and the Lie bracket. Futhermore, the action $\alpha \colon P \times G \to P$ allows us to identify each vertical tangent space $V_u \subseteq \mathrm{T}_u P$ with $\mathfrak{g}$ via the isomorphism
\begin{equation}
  \mathrm{d}\alpha(u, -)_e \colon \mathfrak{g} \to V_u \subseteq \mathrm{T}_u P. 
\end{equation}
